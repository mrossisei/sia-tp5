\section{Autoencoder basico}

\begin{frame}{El problema y el dataset \texttt{font.h}}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \textbf{Dataset}: 32 caracteres de una fuente bitmap.
      \begin{itemize}
        \item Cada letra: grilla \textbf{7$\times$5 = 35 pixeles} binarios $\{0,1\}$.

      \end{itemize}
      \vspace{0.4em}
      \textbf{Pregunta gancho}: ¿se pueden comprimir 35 pixeles a \textbf{2 numeros}
      y reconstruirlos con \textbf{$\le 1$ pixel} de error?
    \end{column}
    \begin{column}{0.46\textwidth}
      \centering
      \textbf{Objetivo}
      \begin{block}{}
        Aprender los \textbf{32 patrones} de 5$\times$7 en un latente de
        \textbf{2 dimensiones} con un error maximo de \textbf{1 pixel incorrecto}.
      \end{block}
      \vspace{0.3em}
      {\footnotesize Ejemplo (letra \texttt{'a'}):}\\[2pt]
      {\ttfamily\footnotesize
        .....\\ ....\#\\ .\#\#\#\#\\ \#...\#\\ \#..\#\#\\ .\#\#.\#\\ .....}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{La respuesta: no linealidad --- arquitectura base del AE}
  Misma idea simetrica con \textbf{cuello 2D}, pero \textbf{ocultas no lineales}:
  \[ 35 \to 60 \to 40 \to 20 \to \mathbf{2} \to 20 \to 40 \to 60 \to 35 \]
  \begin{itemize}
    \item Latente \textbf{lineal} (no satura); salida \textbf{logistica} (pixeles en $(0,1)$).
  \end{itemize}

  \vspace{0.5em}
  \begin{columns}[T]
    \begin{column}{0.32\textwidth}
      \textbf{Optimizadores}
      \begin{itemize}
        \item GD
        \item Momentum
        \item Adam
      \end{itemize}
    \end{column}
    \begin{column}{0.32\textwidth}
      \textbf{Funciones de perdida}
      \begin{itemize}
        \item BCE
        \item MSE
      \end{itemize}
    \end{column}
    \begin{column}{0.32\textwidth}
      \textbf{Activaciones ocultas}
      \begin{itemize}
        \item tanh
        \item ReLU
        \item logistica
      \end{itemize}
    \end{column}
  \end{columns}

  \vspace{0.5em}
  {\footnotesize Esta es la \emph{configuracion base} de referencia: los
   experimentos varian \textbf{una} pieza a la vez.}
\end{frame}

\section{Denoising Autoencoder}

\begin{frame}{Planteo del DAE --- ruido y generalizacion}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      \textbf{Modelos de ruido}
      \begin{itemize}
        \item \textbf{salt\&pepper}: invierte cada pixel con prob. $p$.
        \item \textbf{gaussian}: $x + \mathcal{N}(0,\sigma^2)$, clip a $[0,1]$.
        \item \textbf{masking}: pone a 0 una fraccion de pixeles.
      \end{itemize}
    \end{column}
    \begin{column}{0.44\textwidth}
      \textbf{A tener en cuenta}
      \begin{itemize}
        \item Entrenamos \textbf{solo con salt\&pepper}.
        \item El ruido cambia en cada epoca: \textbf{online augmentation}.

      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Planteo del DAE --- parametros de entrenamiento}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{table}\small
        \begin{tabular}{ll}
          \toprule
          Parametro & Valor \\
          \midrule
          Datos        & \texttt{font.h} ($32\times35$) \\
          Augmentation & online \\
          Arquitectura & $[35,60,40,20,\mathbf{2},20,40,60,35]$ \\
          Activaciones & tanh (oculta), logistica (salida) \\
                       & cuello \emph{lineal} (identidad) \\
          Loss / Opt   & BCE / Adam $10^{-3}$ \\
          Epocas / Batch / Init & 12000 / full-batch / Xavier \\
          Ruido train  & salt\&pepper $p=0.10$ \\
          \textbf{Seeds} & \textbf{[0, 1, 2, 3, 42]} \\
          \bottomrule
        \end{tabular}
      \end{table}
    \end{column}
    \begin{column}{0.44\textwidth}
      \textbf{Objetivo}
      \begin{itemize}
        \item Con una \textbf{entrada ruidosa}, el modelo debe reconstruir el
              \textbf{target limpio}.
      \end{itemize}
      \vspace{0.3em}
      \textbf{¿Por qué online?}
      \begin{itemize}
        \item Para que el DAE no memorice un patron fijo de ruido.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\section{Conclusiones}

\begin{frame}{Conclusiones}
  \begin{itemize}\setlength\itemsep{6pt}
    \item Un AE no lineal con cuello \textbf{2D} memoriza las 32 letras con \textbf{0 píxeles de error},
          .
    \item BCE + tanh + Adam funciona bien y es insensible al lr.
    \item El \textbf{DAE} limpia salt\&pepper hasta $p\approx0.2$--$0.3$ y generaliza a ruidos
          no vistos en train (gaussian, masking).
    \item Cuello \textbf{8D lineal} es óptimo para denoising; el nivel de ruido de train tiene
          un sweet spot en $p\approx0.10$--$0.20$.
    \item \textbf{Límite del AE}: el espacio latente \textbf{no tiene estructura}; muestrear
          puntos arbitrarios no siempre genera letras nuevas. $\Rightarrow$ necesitamos un espacio
          latente organizado.
  \end{itemize}
\end{frame}

\section{Autoencoder Variacional (VAE)}

\begin{frame}{Objetivo e idea del VAE}
  \textbf{Objetivo}: construir un espacio latente \emph{organizado} del que se
  pueda \textbf{muestrear} para generar imagenes nuevas --- lo que el AE basico
  no permite.

  \vspace{0.8em}
  \textbf{Dataset}: emojis $16\!\times\!16$ en escala de grises, 16 clases con
  60 variaciones cada una (960 imagenes).

  \vspace{0.8em}
  \textbf{Plan}: primero \emph{barremos hiperparametros} y luego corremos un VAE
  final con la mejor configuracion encontrada.
\end{frame}

\begin{frame}{Conclusiones EJ2}
  \begin{itemize}\setlength\itemsep{6pt}
    \item El VAE impone $p(z)=\mathcal{N}(0,I)$ via KL $\Rightarrow$ latente
          \textbf{continuo y organizado}: clusters por clase, interpolacion suave.
    \item \textbf{Manifold 2D}: gran parte del latente genera emojis plausibles;
          la grilla $15\!\times\!15$ lo confirma visualmente.
    \item \textbf{Generacion real}: $z\!\sim\!\mathcal{N}(0,I)$
          decodificado produce imagenes reconocibles del dataset.
    \item \textbf{Dim latente $=$ trade-off}: para reconstruccion pura conviene
          \textbf{8D} (la rec satura ahi: 16D ya no aporta), pero elegimos
          \textbf{2D} a proposito --- es lo unico que permite graficar el
          \emph{manifold} y el latente \emph{directamente}, asumiendo el costo en detalle.
    \item \textbf{$\beta=1.0$}: sweet spot rec/KL; KL warmup no es determinante en este caso.
  \end{itemize}
\end{frame}
