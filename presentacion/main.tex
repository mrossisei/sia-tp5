\section{Problema y dataset}
% ===========================================================================

\begin{frame}{El problema}
  \begin{itemize}
    \item Implementar \textbf{Autoencoders} en \textbf{numpy puro}, sin frameworks de DL.
    \item Reusamos la maquinaria de MLP con backpropagation de TP3
          (forward/backward, optimizadores, activaciones, inicializadores).
    \item Lo nuevo del Deep Learning: BCE numericamente estable, inicializacion
          segun la activacion (Glorot/He) y la derivacion completa del VAE.
  \end{itemize}
  \vfill
  \begin{block}{Tres bloques}
    \begin{enumerate}
      \item \textbf{EJ1.a} --- Autoencoder basico (cuello 2D, error $\le 1$ pixel).
      \item \textbf{EJ1.b} --- Denoising Autoencoder.
      \item \textbf{EJ2} --- Variational Autoencoder sobre emojis.
    \end{enumerate}
  \end{block}
\end{frame}

\begin{frame}{Dataset \texttt{font.h}}
  \begin{columns}
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \item 32 caracteres de una fuente de 7 columnas (`` \texttt{`} '' a \texttt{DEL}).
        \item Cada caracter es una grilla \textbf{7$\times$5 = 35 pixeles} binarios $\{0,1\}$.
        \item Se decodifica de hexadecimal a una matriz binaria
              ($X \in \{0,1\}^{32 \times 35}$).
        \item Es un dataset chico y discreto: ideal para forzar al autoencoder a
              \emph{memorizar} a traves de un cuello de botella.
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \centering
      \texttt{Letra 'a':}\\[2pt]
      \texttt{.....}\\
      \texttt{....\#}\\
      \texttt{.\#\#\#\#}\\
      \texttt{\#...\#}\\
      \texttt{\#..\#\#}\\
      \texttt{.\#\#.\#}\\
      \texttt{.....}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Denoising: idea y ruido}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \begin{itemize}
        \item Entrenamos un DAE con \textbf{augmentation online}: cada epoca se

\section{EJ2 --- Variational Autoencoder}
% ===========================================================================

\begin{frame}{Motivacion: los ``agujeros'' del latente}
  \begin{itemize}
    \item Un AE comun aprende un latente \emph{arbitrario}: puntos aislados con
          grandes regiones vacias que decodifican a ruido.
    \item No hay forma principiada de \textbf{muestrear} letras/imagenes nuevas.
  \end{itemize}
  \vfill
  \begin{block}{El VAE: un latente continuo y muestreable}
    \begin{itemize}
      \item El encoder produce una \textbf{distribucion} $q(z|x)=\mathcal{N}(\mu, \sigma^2)$
            en lugar de un punto.
      \item Un termino \textbf{KL} la empuja hacia el prior $\mathcal{N}(0, I)$,
            llenando los agujeros.
      \item Asi podemos generar muestreando $z \sim \mathcal{N}(0,I)$ y decodificando.
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Teoria: ELBO, KL y reparametrizacion}
  Maximizamos el \textbf{ELBO} (equivalente a minimizar):
  \[
    \mathcal{L} = \underbrace{\mathbb{E}_{q(z|x)}\big[-\log p(x|z)\big]}_{\text{reconstruccion (BCE)}}
                 + \beta \cdot \underbrace{D_{\mathrm{KL}}\!\big(q(z|x)\,\|\,p(z)\big)}_{\text{regularizacion}}
  \]
  \vspace{-2pt}
  \begin{itemize}
    \item \textbf{KL cerrado} (prior gaussiano):
          $D_{\mathrm{KL}} = -\tfrac{1}{2}\sum_j \big(1 + \log\sigma_j^2 - \mu_j^2 - \sigma_j^2\big)$.
    \item \textbf{Reparametrizacion} (gradiente a traves del muestreo):
          $z = \mu + \sigma \odot \epsilon$, con $\epsilon \sim \mathcal{N}(0, I)$.
    \item $\beta$ pondera el KL ($\beta$-VAE): controla el trade-off
          reconstruccion vs estructura del latente.
  \end{itemize}
  \vfill
  \begin{alertblock}{Gradient-check}
    Backprop analitico vs diferencias finitas: error relativo maximo
    $\mathbf{1.25\times10^{-10}}$ (tol $10^{-4}$) \textbf{--- PASA}.
  \end{alertblock}
\end{frame}

\section{Conclusiones}

\begin{frame}{Conclusiones}
  \begin{itemize}
    \item \textbf{EJ1.a}: el AE no lineal con cuello \textbf{2D} memoriza las 32
          letras con \textbf{0 pixeles de error} (32/32 exactas), muy por encima
          del baseline PCA 2D ($27\%$ de varianza).
    \item \textbf{EJ1.b}: el DAE limpia ruido entre $\sim$25--250$\times$ mejor que
          el AE basico y \textbf{generaliza a ruido no visto} (gaussiano, masking).
    \item \textbf{EJ2}: el VAE pasa el gradient-check ($1.25\times10^{-10}$),
          aprende un latente \emph{continuo} y permite \textbf{generar} emojis nuevos
          por muestreo e interpolacion.
    \item \textbf{EJ3 (extra, fuera del enunciado)}: el \emph{mismo} VAE portado a
          \textbf{MNIST 28$\times$28}, con un barrido de \textbf{profundidad}
          (2 capas ocultas $\to$ optimo; mas profundo empeora) y de \textbf{dimension
          latente} (satura en 16--32), corrido con un \emph{runner pausable y
          reanudable} (resume bit-identico). Documentado en \texttt{ej3/}.
    \item \textbf{Computo (overnight)}: la grilla de 75 corridas separa fracasos
          reales de censura por presupuesto, y 50000 epocas bajan el rec\_MSE del
          VAE 2D de 4.6 a \textbf{1.3}: varias ``limitaciones'' eran tiempo de
          entrenamiento, no arquitectura. Todos los datos crudos quedan
          guardados para re-analizar.
  \end{itemize}
\end{frame}

\begin{frame}{Limitaciones}
  \begin{itemize}
    \item \textbf{Cuello 2D}: la reconstruccion perfecta del AE basico es
          \emph{memorizacion} de 32 patrones; con datos mas ricos un cuello 2D
          pierde detalle (dim 2 vs dim 8/16 a igual presupuesto). Matiz del
          overnight: parte de esa brecha era \emph{sub-entrenamiento} --- con
          50000 epocas el mismo VAE 2D baja de rec\_MSE 4.6 a 1.3.
    \item El \textbf{AE comun} no tiene un latente muestreable (agujeros);
          el VAE lo resuelve, pero a costa de algo de nitidez (trade-off $\beta$).
    \item Dataset chico y de baja resolucion (font.h 7$\times$5, emojis 16$\times$16):
          buen banco de pruebas pedagogico, no representativo de imagenes naturales.
  \end{itemize}

\end{frame}
